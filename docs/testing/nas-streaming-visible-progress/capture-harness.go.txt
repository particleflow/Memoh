package sessionruntime

import (
	"context"
	"encoding/json"
	"os"
	"testing"

	agentevent "github.com/felinics/memoh/internal/agent/event"
	"github.com/felinics/memoh/internal/agent/runtime/native"
	"github.com/felinics/memoh/internal/agent/turn"
	chatview "github.com/felinics/memoh/internal/agent/view"
)

// Temporary capture harness: calls the actual manager/converter, never a provider.
func TestCaptureContactStreamProtocol(t *testing.T) {
	for _, kind := range []string{"direct", "result"} {
		manager := testRuntimeManager(t, NewMemoryBackend(), "fixture-"+kind)
		sessionID, runID := "stream-session-"+kind, "stream-run-"+kind
		sub, err := manager.Subscribe(context.Background(), "stream-bot", sessionID)
		if err != nil {
			t.Fatal(err)
		}
		defer sub.Close()
		cmd := turn.StartTurnCommand{Query: "Show progress", CurrentChannel: "web"}
		if kind == "result" {
			cmd.TaskResultReceipt = "synthetic-host-result"
			cmd.Query = "MACHINE_CONTEXT_MUST_NOT_RENDER"
		}
		if _, _, err := manager.startRun(context.Background(), runStart{botID: "stream-bot", sessionID: sessionID, runID: runID, turnID: runID, turnPosition: 1, builder: func(context.Context, RunHandle) (RunAdmissionView, error) {
			return RunAdmissionView{RequestUserTurn: chatview.NewRequestUserTurn(cmd, runID)}, nil
		}, abortCh: make(chan struct{}, 1), cancel: func() {}, injectCh: make(chan turn.InjectMessage, 1)}); err != nil {
			t.Fatal(err)
		}
		_ = waitRuntimeEvent(t, sub.C, func(e Event) bool { return e.Delta != nil && e.Delta.CurrentRunView != nil })
		snap, err := manager.Snapshot(context.Background(), "stream-bot", sessionID)
		if err != nil {
			t.Fatal(err)
		}
		frames := []Event{{Type: EventRuntimeSnapshot, SessionID: sessionID, Epoch: snap.Epoch, Seq: snap.Seq, Snapshot: &snap}}
		script := []native.StreamEvent{
			{Type: agentevent.RuntimeStatus, Code: "compacting"},
			{Type: native.EventReasoningDelta, Delta: "Provider reasoning summary"},
			{Type: native.EventTextDelta, Delta: "First streamed fragment"},
			{Type: native.EventTextDelta, Delta: "; second fragment"},
			{Type: agentevent.RuntimeStatus},
			{Type: native.EventAgentEnd, Messages: json.RawMessage(`[{"role":"assistant","content":[{"type":"text","text":"First streamed fragment; second fragment"}]}]`)},
		}
		for _, event := range script {
			if _, err := manager.HandleAgentEvent(context.Background(), requireRunHandle(t, manager, "stream-bot", sessionID, runID), event); err != nil {
				t.Fatal(err)
			}
			frame := waitRuntimeEvent(t, sub.C, func(e Event) bool { return e.Type == EventRuntimeDelta && e.Seq > frames[len(frames)-1].Seq })
			frames = append(frames, frame)
		}
		if _, err := manager.FinishRun(context.Background(), requireRunHandle(t, manager, "stream-bot", sessionID, runID), RunStatusCompleted); err != nil {
			t.Fatal(err)
		}
		frames = append(frames, waitRuntimeEvent(t, sub.C, func(e Event) bool { return e.Type == EventRuntimeDelta && e.Seq > frames[len(frames)-1].Seq }))
		data, err := json.MarshalIndent(frames, "", "  ")
		if err != nil {
			t.Fatal(err)
		}
		if err := os.WriteFile("/workspace/streaming-evidence/"+kind+".json", data, 0600); err != nil {
			t.Fatal(err)
		}
	}
}
