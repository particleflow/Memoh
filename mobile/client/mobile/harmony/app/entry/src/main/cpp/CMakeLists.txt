cmake_minimum_required(VERSION 3.16)
project(begoniaraw LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(RAW_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../.build/raw-codec/LibRaw-0.22.2")
# Use upstream's declared library source set; its *_ph.cpp alternatives must not
# be compiled beside the normal implementation.
file(READ "${RAW_SOURCE}/Makefile.am" RAW_MAKEFILE)
string(FIND "${RAW_MAKEFILE}" "lib_libraw_a_SOURCES =" RAW_BEGIN)
string(FIND "${RAW_MAKEFILE}" "lib_libraw_r_a_CXXFLAGS" RAW_END)
math(EXPR RAW_LENGTH "${RAW_END} - ${RAW_BEGIN}")
string(SUBSTRING "${RAW_MAKEFILE}" ${RAW_BEGIN} ${RAW_LENGTH} RAW_SECTION)
string(REGEX MATCHALL "src/[A-Za-z0-9_/]+\\.cpp" RAW_NAMES "${RAW_SECTION}")
foreach(RAW_NAME IN LISTS RAW_NAMES)
  list(APPEND RAW_FILES "${RAW_SOURCE}/${RAW_NAME}")
endforeach()
add_library(begoniaraw SHARED raw_image.cpp file_uri.cpp ${RAW_FILES})
target_include_directories(begoniaraw PRIVATE "${RAW_SOURCE}")
target_compile_definitions(begoniaraw PRIVATE LIBRAW_NOTHREADS)
target_link_libraries(begoniaraw PUBLIC libace_napi.z.so libohfileuri.so)
